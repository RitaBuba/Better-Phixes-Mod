ig_agrarian_populists = {
	color = hsv{ 0.32 0.67 0.75 }
	texture = "gfx/interface/icons/ig_icons/agrarian_populists.dds"	
	layer = "revolution_dynamic_rural_folk"
	index = 8

	ideologies = {
		ideology_nonexistent
	}

	character_ideologies = {
		ideology_center_leader
		ideology_pacifist
		ideology_protectionist
		ideology_humanitarian
	}
	
	enable = {
		always = yes
	}
	
	on_enable = {
		bpm_update_ideology_agrarian_populists = yes
		bpm_update_name_agrarian_populists = yes
	}
	on_disable = {}

	priority_cultures = { #Country scope
		rule = {
			trigger = {
				c:BIC ?= this
				any_primary_culture = { 
					cu:british = this
				}
			}
			cultures = {
				assamese avadhi baluchi bengali bihari gujarati kannada malayalam marathi oriya panjabi pathan rajput sindi tamil telegu hindustani bundeli pahari gondi bageli chhattisgarhi deccani
			}
		}

		rule = {
			trigger = {
				c:CHI ?= this
				any_primary_culture = { 
					cu:manchu = this
				}
			}
			cultures = {
				han
			}
		}
	}

	on_character_ig_membership = {}
	
	pop_potential = {
		custom_tooltip = {
			text = BPM_VALID_POP_FOR_IG
			owner = {
				ig:ig_agrarian_populists ?= {
					bpm_interest_group_is_activated = yes
				}
			}
		}
		NOR = {
			is_pop_type = capitalists
			is_pop_type = bureaucrats
		}
	}
	
	pop_weight = {
		value = 0

		##### RURAL POP WEIGHTS ONLY #####
		if = {
			limit = { 
				bpm_pop_employment_is_rural = yes
			}
			add = {
				desc = "RURAL_POP"
				value = 50
			}
			# Free farmers are more likely to be agpops
			if = {
				limit = { is_pop_type = farmers }
				add = {
					desc = "POP_FARMERS"
					value = 50
				}
			}
			else_if = {
				limit = { is_pop_type = peasants }
				add = {
					desc = "POP_PEASANTS"
					value = 100
				}
				add = state.modifier:state_bpm_ig_agrarian_populists_peasants_attraction_weight_add
			}
			else_if = {
				limit = { is_pop_type = laborers }
				add = {
					desc = "POP_RURAL_LABORERS"
					value = 50
				}
			}
			else_if = {
				limit = { 
					is_pop_type = aristocrats
				}
				add = {
					desc = "POP_ARISTOCRATS"
					value = -50
				}
			}
			add = state.modifier:state_bpm_ig_agrarian_populists_pro_freedom_attraction_weight_add
			# Literacy adds to agpops
			if = {
				limit = { 
					strata < upper 
				}
				add = {
					desc = "POP_EDUCATED_FARMERS"
					value = literacy_rate
					multiply = 150
					max = 100
				}
			}
			# Agsocs
			if = {
				limit = {
					scope:interest_group = { 
						has_variable = bpm_ig_is_socialist_var 
					}
					strata < upper
				}
				add = state.modifier:state_bpm_ig_agrarian_socialists_attraction_weight_add
				if = {
					limit = {
						owner = { NOT = { has_law_or_variant  = law_type:law_collectivized_agriculture } }						}
					add = {
						desc = "POP_RURAL_WANTS_COLLECTIVIZATION"
						value = 50
					}
				}
			}
		}
		
		##### IF NOT RURAL POP AND NOT ACADEMIC, EXCLUDE #####
		else_if = {
			limit = { NOT = { is_pop_type = academics }  }
			add = {
				desc = "POP_EXCLUSIONS"
				value = -100
			}	
		}

		##### ACADEMICS ARE LEADERS #####
		if = {
			limit = { 
				is_pop_type = academics
			}
			add = state.modifier:state_bpm_ig_agrarian_populists_pro_freedom_attraction_weight_add
		}

		##### PEASANT/FARMER ANGER SYMPATHY SUPPORT #####
		if = {
			limit = { owner = { has_variable = bpm_ig_farmers_angry_var } }
			if = {
				limit = { is_pop_type = soldiers }
				add = {
					desc = "POPS_ANGER_SYMPATHY"
					value = 20
				}
			}
			else_if = {
				limit = {
					OR = {
						is_pop_type = clergymen
						is_pop_type = academics
					}
				}
				add = {
					desc = "POPS_ANGER_SYMPATHY"
					value = 50
				}
			}
		}

		multiply = state.modifier:state_bpm_ig_agrarian_populists_attraction_weight_mult

		multiply = ig_attraction_from_leader_popularity
		multiply = bpm_ig_agitators_popularity
		multiply = bpm_ig_ruler_popularity
		multiply = { desc = bpm_ig_attraction_from_radicals_and_loyalists_tt value = ig_attraction_from_radicals_and_loyalists }
		multiply = ig_attraction_to_ruler_ideology_from_stability
		multiply = ig_attraction_from_government # im guessing this includes the rev one too

	}
	
	monarch_weight = {
		owner.ig:ig_agrarian_populists = {
			value = bpm_ig_monarch_weight
		}
	}	

	agitator_weight = {
		owner.ig:ig_agrarian_populists = {
			value = bpm_base_agitator_weight
		}
	}
	
	commander_weight = {
		owner.ig:ig_agrarian_populists = {
			value = bpm_commander_weight
		}
	}	

	executive_weight = {
		value = 0.1
		if = {
			limit = {
				owner.ig:ig_agrarian_populists = { bpm_interest_group_is_activated = no }
			}
			value = 0.0
		}
	}	
	
	noble_chance = {
		value = 0.2
	}
	
	female_commander_chance = {	
		value = 1.0
		multiply = bpm_female_commander_spawn_chance
	}
	
	female_politician_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance
	}	

	female_agitator_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance		
	}

	female_executive_chance = {
		value = 1.0
		multiply = bpm_female_character_spawn_chance
	}
	
	# The chance that a commander belonging to this IG takes over leadership when it changes
	# scope:character is the most popular commander in the IG
	commander_leader_chance = {
		value = 0.5
		if = {
			limit = {
				exists = scope:character
			}
			multiply = {
				value = 1
				add = scope:character.popularity
				min = 0
			}
		}
	}
}
